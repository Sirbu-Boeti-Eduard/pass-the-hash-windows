% Figure: the two exchange pairs a client performs with the KDC.
% Used in section 2.5.1. Deliberately spare: four arrows, and for each one
% the key that protects it.
\begin{figure}[htbp]
\centering
\small
\begin{tikzpicture}[
  ar/.style={-{Stealth[length=2.2mm]}, draw=black!65, thick},
  msg/.style={align=center, font=\footnotesize, text width=8.4cm},
  key/.style={align=center, font=\scriptsize\itshape, text=black!60},
  life/.style={draw=black!35, thick}
]

\def\cw{0}
\def\kw{9.4}

% lifelines
\draw[life] (\cw,0.55) -- (\cw,-8.1);
\draw[life] (\kw,0.55) -- (\kw,-8.1);
\node[font=\footnotesize\bfseries] at (\cw,0.85) {client};
\node[font=\footnotesize\bfseries] at (\kw,0.85) {KDC \textnormal{(domain controller)}};

% 1 ---------------------------------------------------------------
\draw[ar] (\cw,-0.3) -- (\kw,-0.3)
  node[midway, above=2pt, msg] {\textbf{1\quad AS-REQ}\\
    pre-authentication: a timestamp\\[1pt]
    {\scriptsize\itshape encrypted with the client's long-term key}};
\fill (\cw,-0.3) circle (1.6pt); \fill (\kw,-0.3) circle (1.6pt);

% 2 ---------------------------------------------------------------
\draw[ar] (\kw,-2.7) -- (\cw,-2.7)
  node[midway, above=2pt, msg] {\textbf{2\quad AS-REP}\\
    a session key, and a TGT\\[1pt]
    {\scriptsize\itshape session key under the client key; TGT under the krbtgt key}};
\fill (\cw,-2.7) circle (1.6pt); \fill (\kw,-2.7) circle (1.6pt);

% 3 ---------------------------------------------------------------
\draw[ar] (\cw,-5.1) -- (\kw,-5.1)
  node[midway, above=2pt, msg] {\textbf{3\quad TGS-REQ}\\
    the TGT, an authenticator, and the name of the service wanted\\[1pt]
    {\scriptsize\itshape authenticator under the session key from arrow 2}};
\fill (\cw,-5.1) circle (1.6pt); \fill (\kw,-5.1) circle (1.6pt);

% 4 ---------------------------------------------------------------
\draw[ar] (\kw,-7.5) -- (\cw,-7.5)
  node[midway, above=2pt, msg] {\textbf{4\quad TGS-REP}\\
    a service ticket, and a second session key\\[1pt]
    {\scriptsize\itshape ticket under the service account's key; session key under the first}};
\fill (\cw,-7.5) circle (1.6pt); \fill (\kw,-7.5) circle (1.6pt);

\end{tikzpicture}
\caption{The two exchanges with the KDC, and the key that protects each leg. The long-term key
does its work once, at arrow~1; the rest runs on session keys the KDC hands out, inside tickets
that carry the identity and the rights.}
\label{fig:kerberos-flow}
\end{figure}
