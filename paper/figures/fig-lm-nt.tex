% Figure: the same password reaching two different secrets.
% Used in section 2.2.2, referenced again from 2.3.1.
\begin{figure}[htbp]
\centering
\small
\begin{tikzpicture}[
  box/.style={draw=black!45, rounded corners=2pt, inner sep=4pt, align=center,
              text width=3.0cm, minimum height=1.15cm, font=\footnotesize},
  term/.style={draw=black!70, rounded corners=2pt, inner sep=4pt, align=center,
               text width=3.0cm, minimum height=1.15cm, font=\footnotesize, fill=black!6},
  ar/.style={-{Stealth[length=2mm]}, draw=black!60}
]

% shared source
\node[term] (pw) at (0,-1.7) {\texttt{Password1}\\the only input};

% ---- NT branch (top row) ---------------------------------------------
\node[box] (u16)  at (4.1,0)    {encode UTF-16LE\\18 bytes, case kept};
\node[box] (md4)  at (7.55,0)   {MD4\\single pass, no salt};
\node[term] (nt)  at (11.0,0)   {\texttt{NT hash}\\16 bytes};

% ---- LM branch (bottom row) ------------------------------------------
\node[box] (up)   at (4.1,-3.4) {uppercase, then cut to 14 bytes\\case lost, rest discarded};
\node[box] (des)  at (7.55,-3.4){split 7+7, each half a DES key\\encrypting the constant
                                 \texttt{KGS!@\#\$\%}};
\node[term] (lm)  at (11.0,-3.4){\texttt{LM hash}\\16 bytes, two halves};

% edges
\draw[ar] (pw.east) -- ++(0.5,0) |- (u16.west);
\draw[ar] (pw.east) -- ++(0.5,0) |- (up.west);
\draw[ar] (u16) -- (md4);
\draw[ar] (md4) -- (nt);
\draw[ar] (up) -- (des);
\draw[ar] (des) -- (lm);

% the point
\node[font=\footnotesize, text=black!65, align=left, text width=15.2cm,
      anchor=north west] at (-1.7,-4.9)
  {Same password, two unrelated 16-byte values. The lower path discards case and
   everything past fourteen characters before it hashes anything, so it can only describe a
   fraction of the inputs the upper path can, and it is stored as a weak second record rather
   than as another encoding of the same secret. Neither path salts its input.};

\end{tikzpicture}
\caption{One password, two stored secrets. The upper path condenses the password exactly; the
lower path folds case and length first. Because neither is salted, both values are identical on
every host that holds them for that account.}
\label{fig:lm-nt}
\end{figure}
