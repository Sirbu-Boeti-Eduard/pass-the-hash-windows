% Figure: one password, the two hashes a Windows record stores for it.
% The LM step is one box holding two key rows: seven bytes of password plus the
% null byte that brings each to the eight a DES key takes. The DES box names the
% key in the subscript, so the password chunk is visibly the key rather than the
% plaintext. Values for Password1: NT 5bc6..., LM 93fab.../060799...
\begin{figure}[H]
\centering
\captionsetup{skip=5pt}
\begin{tikzpicture}[
  box/.style={draw=black!45, rounded corners=2pt, inner sep=4pt, align=center,
              font=\footnotesize, minimum height=1.0cm},
  term/.style={draw=black!70, rounded corners=2pt, inner sep=4pt, align=center,
               font=\footnotesize, fill=black!6, minimum height=1.0cm},
  sub/.style={draw=black!30, rounded corners=1.5pt, inner sep=1.5pt, align=left},
  ar/.style={-{Stealth[length=2mm]}, draw=black!60}
]

% ---------------- LM step: uppercase, cut, two keys (one big box) ----------
\node[align=center, font=\scriptsize, anchor=north] (lup) at (4.7,-2.20)
  {uppercase, {\footnotesize\texttt{PASSWORD1}}, cut to 14};
\node[sub, anchor=north] (k1) at ([yshift=-5pt]lup.south)
  {\textcolor{black!60}{\scriptsize key 1}\enspace{\scriptsize\texttt{50415353574f52}}\,{\scriptsize\texttt{+}}\textcolor{black!45}{\scriptsize\texttt{00}}};
\node[sub, anchor=north] (k2) at ([yshift=-2pt]k1.south)
  {\textcolor{black!60}{\scriptsize key 2}\enspace{\scriptsize\texttt{44310000000000}}\,{\scriptsize\texttt{+}}\textcolor{black!45}{\scriptsize\texttt{00}}};
\node[box, fit=(lup)(k1)(k2), inner sep=4pt] (step) {};

% ---------------- DES and its two results ---------------------------------
\node[box, right=0.55cm of step] (des)
  {{\scriptsize$\mathtt{DES}_{k_1}(\mathtt{KGS!@\#\$\%})$}\\[1pt]
   {\scriptsize$\mathtt{DES}_{k_2}(\mathtt{KGS!@\#\$\%})$}};

% ---------------- stored values -------------------------------------------
\node[term, right=0.55cm of des] (lmh)
  {\texttt{LM hash, 16 bytes}\\[1pt]
   {\scriptsize\texttt{93fab6c61519a5e5}}\\
   {\scriptsize\texttt{060799828bfccceb}}};

\node[box] (md4) at (step |- 0,0) {MD4};
\node[term] (nth) at (lmh |- 0,0)
  {\texttt{NT hash, 16 bytes}\\[1pt]
   {\scriptsize\texttt{5bc6fc211d85ee6e}}\\
   {\scriptsize\texttt{186790b1272c942d}}};

% ---------------- one password, two lanes ---------------------------------
\node[term] (pw) at (0.6,-1.72) {\texttt{Password1}};

\draw[ar] (pw.east) -- ++(0.4,0) coordinate (fork);
\draw[ar] (fork) |- (md4.west);
\draw[ar] (fork) |- (step.west);
\draw[ar] (md4) -- (nth);
\draw[ar] (step) -- (des);
\draw[ar] (des) -- (lmh);

\end{tikzpicture}
\caption{One password, two stored values: the NT hash and the LM hash.}
\label{fig:lm-nt}
\end{figure}
